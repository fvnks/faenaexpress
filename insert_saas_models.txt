// FACTURACIÓN SAAS: facturación de la plataforma a las organizaciones
//
// Son facturas que EMITE la plataforma (Yellow ERP) a las cuentas, no las
// que las cuentas emiten a sus clientes (eso está en Sale/Purchase).
// Mantiene su propio ciclo de vida y su propia tabla de pagos.
model PlatformInvoice {
  id            String   @id @default(cuid())
  organizationId String
  // Número correlativo por organización (F-2026-0001, F-2026-0002...)
  number        String
  period        String   // YYYY-MM
  status        PlatformInvoiceStatus @default(PENDING)
  // En centavos (CLP)
  amountCLP     Int
  taxCLP        Int      @default(0)
  totalCLP      Int
  currency      String   @default("CLP")
  // Qué cubre: descripción del período/plana
  description   String
  // Si fue emitida vía SII (tiene trackId, folio, etc.)
  siiTrackId    String?
  siiFolio      Int?
  siiStatus     String?
  // Vencimiento y pagos
  dueAt         DateTime
  paidAt        DateTime?
  // Metadatos del plan que originó esta factura
  planCode      String
  planName      String
  planPriceCLP  Int
  createdAt     DateTime @default(now())
  updatedAt     DateTime @updatedAt

  organization  Organization @relation(fields: [organizationId], references: [id], onDelete: Cascade)
  payments      PlatformPayment[]

  @@unique([organizationId, number])
  @@index([organizationId, status])
  @@index([organizationId, period])
  @@index([dueAt])
  @@index([siiTrackId])
}

model PlatformPayment {
  id            String   @id @default(cuid())
  invoiceId     String
  organizationId String
  // Método: FLOW, MERCADOPAGO, TRANSFER, MANUAL
  provider      PaymentProvider
  providerRef   String?  // ID de la transacción en el proveedor
  // En centavos
  amountCLP     Int
  currency      String   @default("CLP")
  status        PlatformPaymentStatus @default(PENDING)
  // Respuesta cruda del proveedor (JSON string)
  providerData  String?
  paidAt        DateTime?
  createdAt     DateTime @default(now())
  updatedAt     DateTime @updatedAt

  invoice       PlatformInvoice @relation(fields: [invoiceId], references: [id], onDelete: Cascade)

  @@index([invoiceId])
  @@index([organizationId])
  @@index([provider, providerRef])
  @@index([status])
}

model PlatformPaymentMethod {
  id            String   @id @default(cuid())
  organizationId String
  // Método guardado: FLOW, MERCADOPAGO, TRANSFER
  provider      PaymentProvider
  // Token/identificador en el proveedor (para pagos recurrentes)
  providerToken String?
  // Datos visibles para el usuario (últimos 4 dígitos, banco, etc.)
  displayName   String
  isDefault     Boolean  @default(false)
  active        Boolean  @default(true)
  createdAt     DateTime @default(now())
  updatedAt     DateTime @updatedAt

  organization  Organization @relation(fields: [organizationId], references: [id], onDelete: Cascade)

  @@unique([organizationId, provider, providerToken])
  @@index([organizationId])
}

enum PlatformInvoiceStatus {
  DRAFT       // Borrador (no enviada)
  PENDING     // Emitida, pendiente de pago
  PARTIAL     // Pago parcial
  PAID        // Pagada
  VOID        // Anulada
  OVERDUE     // Vencida sin pagar (morosa)
}

enum PlatformPaymentStatus {
  PENDING
  PROCESSING
  COMPLETED
  FAILED
  REFUNDED
  CANCELED
}

enum PaymentProvider {
  FLOW
  MERCADOPAGO
  TRANSFER
  MANUAL
  CARD
}
